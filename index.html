🛡️ Firestore Security RulesDeploy Ready

Copy these rules into your Firebase Console → Firestore → Rules tab to enforce server-side RBAC.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isSignedIn()     { return request.auth != null; }
    function userRole()       { return get(/databases/$(database)/documents/userRoles/$(request.auth.uid)).data.role; }
    function isSuperAdmin()   { return isSignedIn() && userRole() == 'super_admin'; }
    function isHRManager()    { return isSignedIn() && userRole() in ['super_admin','hr_manager']; }
    function isPayrollAdmin() { return isSignedIn() && userRole() in ['super_admin','hr_manager','payroll_admin']; }
    function isManager()      { return isSignedIn() && userRole() in ['super_admin','hr_manager','manager']; }
    function isOwner(uid)     { return isSignedIn() && request.auth.uid == uid; }

    match /employees/{empId} {
      allow read:   if isManager() || isOwner(resource.data.uid);
      allow create: if isHRManager();
      allow update: if isHRManager();
      allow delete: if isSuperAdmin();
    }
    match /payroll/{docId} {
      allow read:   if isPayrollAdmin() || isOwner(resource.data.uid);
      allow write:  if isPayrollAdmin();
    }
    match /leaves/{leaveId} {
      allow read:   if isManager() || isOwner(resource.data.uid);
      allow create: if isSignedIn();
      allow update: if isManager();
      allow delete: if isHRManager();
    }
    match /auditLogs/{logId} {
      allow create: if isSignedIn();
      allow read:   if isHRManager();
      allow update, delete: if false;
    }
    match /userRoles/{uid} {
      allow read:   if isSignedIn() && (isSuperAdmin() || isOwner(uid));
      allow write:  if isSuperAdmin();
    }
  }
}
